cmake_minimum_required(VERSION 3.22)
project(framesink LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Qml Quick)
qt_add_qml_module(framesink URI org.forge.framesink VERSION 1.0 SOURCES framesink.cpp framesink.h)
target_link_libraries(framesink PRIVATE Qt6::Quick Qt6::Qml)
